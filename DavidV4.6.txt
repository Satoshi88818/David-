""" dreamer.py v4.6 — Pinocchio-Based Stability & Multi-Hypothesis Dreaming Improvements over v4.5: - Real centroidal dynamics via Pinocchio (with graceful fallback to heuristic) - Multi-hypothesis scoring across several candidate primitives - Energy-based stability metric - Distance-to-stability-boundary reporting """ import numpy as np import logging from typing import List, Tuple from cognitive_core.schemas import DreamerResult, HumanoidMotorPrimitive logger = logging.getLogger(__name__) try: import pinocchio as pin HAS_PINOCCHIO = True except ImportError: HAS_PINOCCHIO = False logger.warning("[Dreamer] Pinocchio not available — using heuristic stability fallback.") class SupportPolygon: """Axis-aligned approximation of bipedal support polygon.""" def __init__(self, half_width: float = 0.12, half_length: float = 0.15): self.half_width = half_width self.half_length = half_length def contains(self, com_xy: np.ndarray) -> bool: return (abs(com_xy[0]) < self.half_length and abs(com_xy[1]) < self.half_width) def distance_to_boundary(self, com_xy: np.ndarray) -> float: """Returns distance to nearest boundary edge (negative if outside).""" dx = self.half_length - abs(com_xy[0]) dy = self.half_width - abs(com_xy[1]) return min(dx, dy) class DreamerLoop: """ Simulates full-body movements and scores their safety. Usage: result = dreamer.simulate(depth_frame, primitive, world_model) best = dreamer.score_candidates(depth_frame, candidates, world_model) """ def __init__(self, urdf_path: str = ""): self.support_polygon = SupportPolygon() self.pin_model = None self.pin_data = None if HAS_PINOCCHIO and urdf_path: try: self.pin_model = pin.buildModelFromUrdf(urdf_path) self.pin_data = self.pin_model.createData() logger.info("[Dreamer] Pinocchio model loaded from URDF.") except Exception as e: logger.warning(f"[Dreamer] URDF load failed: {e}. Falling back to heuristics.") # ------------------------------------------------------------------ # Public API # ------------------------------------------------------------------ def simulate( self, current_depth: np.ndarray, proposed_primitive: HumanoidMotorPrimitive, world_model, ) -> DreamerResult: """Simulate a single primitive and return safety metrics.""" collision_prob = self._check_collisions(world_model, proposed_primitive) fall_risk, stability_margin = self._check_stability(proposed_primitive, world_model) energy = self._estimate_energy(proposed_primitive) safe_base = None if fall_risk > 0.4: logger.warning(f"[Dreamer] Fall risk={fall_risk:.2f}. Suggesting base adjustment.") safe_base = self._compute_safe_base_step(proposed_primitive, world_model) return DreamerResult( simulated=True, collision_prob=collision_prob, fall_risk=fall_risk, stability_margin=stability_margin, energy_cost_estimate=energy, safe_base_adjustment=safe_base, ) def score_candidates( self, current_depth: np.ndarray, candidates: List[HumanoidMotorPrimitive], world_model, ) -> Tuple[HumanoidMotorPrimitive, DreamerResult]: """ Score multiple candidate primitives and return the safest one. Returns (best_primitive, best_result). """ best_prim = candidates[0] best_result = self.simulate(current_depth, candidates[0], world_model) best_score = self._composite_score(best_result) for prim in candidates[1:]: result = self.simulate(current_depth, prim, world_model) score = self._composite_score(result) if score < best_score: # lower = safer best_score = score best_prim = prim best_result = result logger.info( f"[Dreamer] Best candidate: '{best_prim.primitive_name}' " f"(score={best_score:.3f}, fall_risk={best_result.fall_risk:.2f})" ) return best_prim, best_result # ------------------------------------------------------------------ # Internal helpers # ------------------------------------------------------------------ def _check_stability( self, primitive: HumanoidMotorPrimitive, world_model ) -> Tuple[float, float]: """ Returns (fall_risk, stability_margin). Uses Pinocchio centroidal dynamics when available, heuristic otherwise. """ target = primitive.right_hand_xyz or primitive.left_hand_xyz if target is None: return 0.0, self.support_polygon.half_length if self.pin_model is not None: return self._pinocchio_stability(primitive, world_model, target) else: return self._heuristic_stability(target) def _pinocchio_stability( self, primitive: HumanoidMotorPrimitive, world_model, target: list, ) -> Tuple[float, float]: """Compute projected CoM using Pinocchio forward kinematics.""" try: q = np.array(world_model.current_joint_config) # Apply a virtual joint displacement toward target delta_q = np.zeros_like(q) if len(target) >= 3: delta_q[0] = np.clip(target[0] * 0.1, -0.05, 0.05) delta_q[1] = np.clip(target[1] * 0.1, -0.05, 0.05) q_virtual = q + delta_q pin.forwardKinematics(self.pin_model, self.pin_data, q_virtual) pin.centerOfMass(self.pin_model, self.pin_data, q_virtual) com = self.pin_data.com[0][:2] # x, y projection margin = self.support_polygon.distance_to_boundary(com) if margin < 0: fall_risk = min(0.95, 0.85 + abs(margin) * 2.0) elif margin < 0.03: fall_risk = 0.6 elif margin < 0.07: fall_risk = 0.3 else: fall_risk = 0.1 return fall_risk, max(margin, 0.0) except Exception as e: logger.warning(f"[Dreamer] Pinocchio computation failed: {e}. Using heuristic.") return self._heuristic_stability(target) def _heuristic_stability(self, target: list) -> Tuple[float, float]: """Simple distance-from-centre heuristic (v4.5 fallback).""" dist = np.linalg.norm([target[0], target[1]]) if dist > 0.7: return 0.9, 0.0 if dist > 0.5: return 0.5, 0.05 return 0.1, 0.12 def _check_collisions(self, world_model, primitive: HumanoidMotorPrimitive) -> float: """Voxel sweep collision check (placeholder — integrate with OctoMap/world_model).""" # TODO: integrate with world_model.occupancy_grid return 0.0 def _estimate_energy(self, primitive: HumanoidMotorPrimitive) -> float: """Heuristic energy estimate based on locomotion mode and reach distance.""" base = 0.1 if primitive.locomotion_mode in ("walk", "crouch"): base += 0.3 target = primitive.right_hand_xyz or primitive.left_hand_xyz if target: reach = np.linalg.norm(target) base += np.clip(reach / 1.5, 0.0, 0.5) return float(np.clip(base, 0.0, 1.0)) def _composite_score(self, result: DreamerResult) -> float: """Lower is better. Weighted combination of risk metrics.""" return ( 0.5 * result.fall_risk + 0.3 * result.collision_prob + 0.2 * result.energy_cost_estimate ) def _compute_safe_base_step( self, primitive: HumanoidMotorPrimitive, world_model ) -> list[float]: """Suggest a base movement that reduces fall risk.""" target = primitive.right_hand_xyz or primitive.left_hand_xyz if target is None: return [0.0, 0.0, 0.0] direction = np.array(target[:2]) norm = np.linalg.norm(direction) if norm < 1e-6: return [0.1, 0.0, 0.0] unit = direction / norm step = unit * 0.15 # 15 cm step toward target return [float(step[0]), float(step[1]), 0.0]
