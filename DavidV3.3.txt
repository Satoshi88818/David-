Architecture (v3.3 — Improvement #1): 1. At init, embed every skill description once. 2. _resolve(task) encodes the task and finds the nearest skill by cosine sim. 3. 6. cognitive_core/graph_memory.py (NEW)

""" graph_memory.py — Causal / Graph Memory layer for David v3.3. Architecture (Improvement #3 — Graph Memory): - NetworkX directed graph where nodes are memory episodes and edges carry causal relationships (precondition → outcome, entity links). - Supports schema induction by finding recurring path patterns. - Counterfactual imagination: traverse graph to ask "what if preconditions were met?" - Serialised to disk as GraphML for persistence across sessions. Node attributes: text, timestamp, importance, surprise_score, memory_type, session_id Edge attributes: relation: "caused_by" | "precondition_of" | "entity_link" | "schema_instance" weight: float [0, 1] """ from __future__ import annotations import json import logging import os import time from pathlib import Path import networkx as nx logger = logging.getLogger(__name__) _GRAPH_FILE = "causal_graph.graphml" _MAX_GRAPH_NODES = 5_000 # prune when exceeded class GraphMemory: """ Persistent causal graph memory backed by NetworkX. Lives alongside ChromaDB — vector store handles fuzzy retrieval, graph handles causal reasoning and schema induction. """ def __init__(self, persist_dir: str): self._path = Path(persist_dir) / _GRAPH_FILE self._graph: nx.DiGraph = nx.DiGraph() def load(self): """Load existing graph from disk or start fresh.""" if self._path.exists(): try: self._graph = nx.read_graphml(str(self._path)) logger.info( f"GraphMemory loaded: {self._graph.number_of_nodes()} nodes, " f"{self._graph.number_of_edges()} edges." ) except Exception as exc: logger.error(f"GraphMemory load error: {exc}; starting fresh.") self._graph = nx.DiGraph() else: logger.info("GraphMemory: no existing graph, starting fresh.") def save(self): """Persist graph to disk.""" try: self._path.parent.mkdir(parents=True, exist_ok=True) nx.write_graphml(self._graph, str(self._path)) except Exception as exc: logger.error(f"GraphMemory save error: {exc}") # ------------------------------------------------------------------ # Node management # ------------------------------------------------------------------ def add_episode( self, node_id: str, text: str, memory_type: str = "episodic", importance: float = 1.0, surprise_score: float = 0.0, success: bool = True, reflex_triggered: bool = False, session_id: str = "", ): self._graph.add_node( node_id, text=text[:300], memory_type=memory_type, importance=importance, surprise_score=surprise_score, success=str(success), reflex_triggered=str(reflex_triggered), session_id=session_id, timestamp=str(time.time()), ) if self._graph.number_of_nodes() > _MAX_GRAPH_NODES: self._prune() def link( self, from_id: str, to_id: str, relation: str = "caused_by", weight: float = 1.0, ): """Add a directed causal edge between two memory nodes.""" if from_id in self._graph and to_id in self._graph: self._graph.add_edge(from_id, to_id, relation=relation, weight=weight) # ------------------------------------------------------------------ # Causal reasoning # ------------------------------------------------------------------ def get_causes(self, node_id: str, depth: int = 2) -> list[dict]: """Return antecedent nodes up to `depth` hops back.""" results = [] try: for anc in nx.ancestors(self._graph, node_id): path_len = nx.shortest_path_length(self._graph, anc, node_id) if path_len <= depth: data = dict(self._graph.nodes[anc]) data["node_id"] = anc data["hops"] = path_len results.append(data) except nx.NetworkXError: pass return sorted(results, key=lambda x: x["hops"]) def get_outcomes(self, node_id: str, depth: int = 2) -> list[dict]: """Return descendant nodes (what followed from this episode).""" results = [] try: for desc in nx.descendants(self._graph, node_id): path_len = nx.shortest_path_length(self._graph, node_id, desc) if path_len <= depth: data = dict(self._graph.nodes[desc]) data["node_id"] = desc data["hops"] = path_len results.append(data) except nx.NetworkXError: pass return sorted(results, key=lambda x: x["hops"]) def counterfactual_paths(self, failure_node_id: str) -> list[str]: """ Imagine: what alternative predecessors exist that did NOT lead to the same failure? Returns their text as hypotheticals. """ if failure_node_id not in self._graph: return [] causes = self.get_causes(failure_node_id, depth=1) alternatives: list[str] = [] for cause in causes: # Find sibling nodes (same predecessors, different successors) nid = cause["node_id"] siblings = list(self._graph.successors(nid)) for sib in siblings: if sib == failure_node_id: continue sib_data = self._graph.nodes.get(sib, {}) if sib_data.get("success", "True") == "True": alternatives.append( f"[COUNTERFACTUAL] Instead of '{cause['text'][:60]}', " f"'{sib_data.get('text', '')[:60]}' led to success." ) return alternatives # ------------------------------------------------------------------ # Schema induction via graph patterns # ------------------------------------------------------------------ def find_recurring_patterns(self, min_frequency: int = 3) -> list[str]: """ Find pairs of node types that recur frequently as edges. Returns human-readable schema strings. """ edge_texts: dict[tuple, int] = {} for u, v, data in self._graph.edges(data=True): u_data = self._graph.nodes.get(u, {}) v_data = self._graph.nodes.get(v, {}) key = ( u_data.get("text", "")[:50], data.get("relation", ""), v_data.get("text", "")[:50], ) edge_texts[key] = edge_texts.get(key, 0) + 1 schemas = [] for (u_text, rel, v_text), count in edge_texts.items(): if count >= min_frequency: schemas.append( f"[GRAPH-SCHEMA] '{u_text}' --{rel}--> '{v_text}' " f"(observed {count}x)" ) return schemas # ------------------------------------------------------------------ # Relevance update (active forgetting + feedback) # ------------------------------------------------------------------ def update_importance(self, node_id: str, importance: float): if node_id in self._graph: self._graph.nodes[node_id]["importance"] = importance def apply_ebbinghaus_decay(self, half_life_days: float = 30.0): """ Apply time-based importance decay to all episodic nodes. Nodes below 0.05 importance are marked for pruning. """ now = time.time() to_prune: list[str] = [] for nid, data in self._graph.nodes(data=True): if data.get("memory_type", "") != "episodic": continue try: ts = float(data.get("timestamp", now)) except (ValueError, TypeError): ts = now age_days = (now - ts) / 86400.0 decay = 0.5 ** (age_days / half_life_days) try: imp = float(data.get("importance", 1.0)) except (ValueError, TypeError): imp = 1.0 new_imp = imp * decay self._graph.nodes[nid]["importance"] = new_imp if new_imp < 0.05: to_prune.append(nid) if to_prune: self._graph.remove_nodes_from(to_prune) logger.info(f"GraphMemory: decayed and pruned {len(to_prune)} nodes.") # ------------------------------------------------------------------ # Prune oldest low-importance nodes when graph is too large # ------------------------------------------------------------------ def _prune(self): nodes_by_importance = sorted( self._graph.nodes(data=True), key=lambda x: float(x[1].get("importance", 1.0)), ) to_remove = [nid for nid, _ in nodes_by_importance[: _MAX_GRAPH_NODES // 5]] self._graph.remove_nodes_from(to_remove) logger.info(f"GraphMemory: pruned {len(to_remove)} low-importance nodes.") # ------------------------------------------------------------------ # Stats # ------------------------------------------------------------------ def stats(self) -> dict: return { "nodes": self._graph.number_of_nodes(), "edges": self._graph.number_of_edges(), }

""" MemoryStore — Multi-collection vector store backed by ChromaDB (v3.3).
