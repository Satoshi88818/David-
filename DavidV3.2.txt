""" WorldModel — dynamic representation of the robot's environment. v3.2 changes (Improvement #2 — Semantic Proprioception): - update_proprioception() receives pre-digested semantic strings from StemStateNode. - Stores the latest proprioception_event for context injection into reflection prompts. - Detects vision/proprioception contradictions and boosts surprise accordingly. """ from __future__ import annotations import time from dataclasses import dataclass, field from typing import TYPE_CHECKING import numpy as np if TYPE_CHECKING: from cognitive_core import CognitiveCore _CENTROID_WINDOW = 50 SURPRISE_THRESHOLD = 0.30 @dataclass class WorldModel: current_perception: str = "" active_task: str = "" active_goal: str = "" known_surfaces: set = field(default_factory=set) known_objects: set = field(default_factory=set) recent_failure_contexts: list = field(default_factory=list) last_updated: float = field(default_factory=time.time) # v3.2: latest semantic proprioception string from StemStateNode proprioception_event: str = "" # e.g. "status: slipping, friction_low" proprioception_status: str = "stable" # parsed high-level status token _perception_embeddings: list = field(default_factory=list, repr=False) _centroid: object = field(default=None, repr=False) # ------------------------------------------------------------------ # Perception update (vision) # ------------------------------------------------------------------ def update_perception(self, caption: str, embedding: list[float] | None = None): self.current_perception = caption self.last_updated = time.time() for word in caption.lower().split(): if word in {"smooth", "rough", "metal", "plastic", "wood", "glass", "fabric"}: self.known_surfaces.add(word) if embedding is not None: vec = np.array(embedding, dtype=np.float32) self._perception_embeddings.append(vec) if len(self._perception_embeddings) > _CENTROID_WINDOW: self._perception_embeddings.pop(0) self._centroid = None # ------------------------------------------------------------------ # Proprioception update (v3.2 — Improvement #2) # ------------------------------------------------------------------ def update_proprioception(self, event: str): """ Receive a semantic event string from StemStateNode. 8. cognitive_core/dreamer.py

Changes from v3.1: _experience_replay() and _failure_analysis() now grant an instant massive importance boost when reflex_triggered=True is set in memory metadata, because a reflex firing means physical safety was compromised.

""" DreamEngine — offline consolidation engine for David v3.2. v3.2 changes (Improvement #3 — Reflex Arc integration): - reflex_triggered memories get a REFLEX_IMPORTANCE_MULTIPLIER boost during replay. - This ensures the Brain reflects deeply on any event that forced a physical override.
