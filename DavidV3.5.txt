4. cognitive_core/memory_store.py

""" memory_store.py — David v3.5 Changes from v3.4: - Surprise-gated flywheel (Improvement 4): * surprise > 0.7 OR success == False → save full-res frame * success == True AND surprise < 0.2 → discard (or 10% random save) * Otherwise (0.2–0.7) → save at 50% probability This creates a Hard Example Mining dataset for fine-tuning. """ from __future__ import annotations import base64 import logging import os import random import time import uuid from typing import Optional import chromadb from chromadb.utils import embedding_functions from cognitive_core.schemas import MemoryType logger = logging.getLogger(__name__) # ── Surprise thresholds ────────────────────────────────────────────────────── SURPRISE_ALWAYS_SAVE = 0.7 # Save unconditionally above this SURPRISE_NEVER_SAVE = 0.2 # Save only at ROUTINE_SAVE_PROB below this ROUTINE_SAVE_PROB = 0.10 # ── Core write API ──────────────────────────────────────────────────────── async def add( self, text: str, memory_type: MemoryType = MemoryType.EPISODIC, metadata: dict | None = None, location: dict | None = None, session_id: str | None = None, parent_id: str | None = None, image_b64: str | None = None, ) -> str: mem_id = str(uuid.uuid4()) meta: dict = { "timestamp": time.time(), "importance": 1.0, "success": True, "surprise_score": 0.0, **(metadata or {}), } if location: meta["location"] = str(location) if session_id: meta["session_id"] = session_id if parent_id: meta["parent_id"] = parent_id # ── Surprise-gated flywheel (Improvement 4) ───────────────────────── if image_b64 and memory_type == MemoryType.EPISODIC: surprise = float(meta.get("surprise_score", 0.0)) success = bool(meta.get("success", True)) should_save = self._flywheel_gate(surprise, success) if should_save: try: frame_path = os.path.join(self._frames_dir, f"{mem_id}.jpg") with open(frame_path, "wb") as f: f.write(base64.b64decode(image_b64)) meta["image_path"] = frame_path logger.debug( f"Flywheel saved frame {mem_id}.jpg " f"(surprise={surprise:.2f}, success={success})" ) except Exception as e: logger.error(f"Flywheel frame save failed: {e}") else: logger.debug( f"Flywheel discarded routine frame " f"(surprise={surprise:.2f}, success={success})" ) #
