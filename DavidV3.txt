""" cognitive_core — self-contained local cognition package for David v3.0. Exposes the CognitiveCore facade that all ROS nodes depend on. """ from .memory_store import MemoryStore, MemoryType from .embedder import Embedder from .dreamer import DreamEngine from .world_model import WorldModel from .reflection import ReflectionEngine import asyncio import logging logger = logging.getLogger(__name__) class CognitiveCore: """ Drop-in replacement for MemoriaClient. All operations are local — no network required. """ def __init__(self, persist_dir: str = "/var/david/memory"): self.embedder = Embedder() self.memory = MemoryStore(persist_dir=persist_dir, embedder=self.embedder) self.world_model = WorldModel() self.dreamer = DreamEngine(memory=self.memory, embedder=self.embedder) self.reflection = ReflectionEngine(memory=self.memory) async def initialise(self): """Load persisted collections and warm up the embedder.""" await self.memory.initialise() self.embedder.warmup() logger.info("CognitiveCore initialised (local).") # ------------------------------------------------------------------ # Primary interface (matches MemoriaClient surface) # ------------------------------------------------------------------ async def store( self, text: str, memory_type: MemoryType = MemoryType.EPISODIC, metadata: dict | None = None, ): await self.memory.add(text=text, memory_type=memory_type, metadata=metadata or {}) async def retrieve( self, query: str, k: int = 8, memory_type: MemoryType | None = None, filters: dict | None = None, ) -> list[dict]: return await self.memory.search( query=query, k=k, memory_type=memory_type, filters=filters ) async def consolidate(self): """Run the full dream cycle (call only when docked).""" await self.dreamer.run_full_cycle() async def reflect(self, context: dict) -> str: """Ask 'why did this fail / succeed?' and update procedural memory.""" return await self.reflection.analyse(context) async def generate_goal(self) -> str | None: """Return a curiosity-driven exploration goal, or None if none found.""" return await self.reflection.curiosity_goal(self.world_model) async def abort_consolidation(self): """Interrupt an in-progress dream cycle immediately.""" await self.dreamer.abort() 

4. cognitive_core/embedder.py

""" Local embedding model wrapper. Uses sentence-transformers/all-MiniLM-L6-v2 by default (fast, ~80 MB). Falls back gracefully if the model is not yet downloaded. """ from __future__ import annotations import logging import numpy as np from sentence_transformers import SentenceTransformer logger = logging.getLogger(__name__) _MODEL_NAME = "sentence-transformers/all-MiniLM-L6-v2" class Embedder: def __init__(self, model_name: str = _MODEL_NAME): self._model_name = model_name self._model: SentenceTransformer | None = None def warmup(self): """Eagerly load the model so first inference is fast.""" logger.info(f"Loading embedding model: {self._model_name}") self._model = SentenceTransformer(self._model_name) logger.info("Embedding model loaded.") def _get_model(self) -> SentenceTransformer: if self._model is None: self.warmup() return self._model # type: ignore[return-value] def embed(self, text: str) -> list[float]: """Return a unit-normalised embedding vector.""" model = self._get_model() vec: np.ndarray = model.encode(text, normalize_embeddings=True) return vec.tolist() def embed_batch(self, texts: list[str]) -> list[list[float]]: model = self._get_model() vecs: np.ndarray = model.encode(texts, normalize_embeddings=True, batch_size=32) return vecs.tolist() 

5. cognitive_core/memory_store.py

""" Multi-collection vector store backed by ChromaDB (local, persistent). Memory types: WORKING — short-lived in-context window (auto-expires after N entries) EPISODIC — what happened (raw observations + outcomes) SEMANTIC — derived facts, schemas, generalised knowledge PROCEDURAL — skills, preconditions, failure patterns """ from __future__ import annotations import asyncio import logging import time import uuid from enum import Enum import chromadb from chromadb.config import Settings from .embedder import Embedder logger = logging.getLogger(__name__) WORKING_MEMORY_LIMIT = 20 # Max entries before oldest is evicted DECAY_HALF_LIFE_DAYS = 30.0 # Importance decay for episodic memories class MemoryType(str, Enum): WORKING = "working" EPISODIC = "episodic" SEMANTIC = "semantic" PROCEDURAL = "procedural" class MemoryStore: def __init__(self, persist_dir: str, embedder: Embedder): self._dir = persist_dir self._embedder = embedder self._client: chromadb.PersistentClient | None = None self._collections: dict[MemoryType, chromadb.Collection] = {} async def initialise(self): loop = asyncio.get_event_loop() await loop.run_in_executor(None, self._sync_init) def _sync_init(self): self._client = chromadb.PersistentClient( path=self._dir, settings=Settings(anonymized_telemetry=False), ) for mt in MemoryType: self._collections[mt] = self._client.get_or_create_collection( name=mt.value, metadata={"hnsw:space": "cosine"}, ) logger.info(f"MemoryStore initialised at {self._dir}") # ------------------------------------------------------------------ # Write # ------------------------------------------------------------------ async def add( self, text: str, memory_type: MemoryType = MemoryType.EPISODIC, metadata: dict | None = None, ) -> str: mem_id = str(uuid.uuid4()) meta = { "timestamp": time.time(), "importance": 1.0, "success": metadata.get("success", True) if metadata else True, **(metadata or {}), } embedding = self._embedder.embed(text) col = self._collections[memory_type] loop = asyncio.get_event_loop() await loop.run_in_executor( None, lambda: col.add( ids=[mem_id], documents=[text], embeddings=[embedding], metadatas=[meta], ), ) # Enforce working memory limit if memory_type == MemoryType.WORKING: await self._evict_oldest_working() return mem_id async def _evict_oldest_working(self): col = self._collections[MemoryType.WORKING] loop = asyncio.get_event_loop() result = await loop.run_in_executor(None, lambda: col.get(include=["metadatas"])) ids = result["ids"] if len(ids) > WORKING_MEMORY_LIMIT: metas = result["metadatas"] paired = sorted(zip(metas, ids), key=lambda x: x[0]["timestamp"]) to_delete = [iid for _, iid in paired[: len(ids) - WORKING_MEMORY_LIMIT]] await loop.run_in_executor(None, lambda: col.delete(ids=to_delete)) # ------------------------------------------------------------------ # Read # ------------------------------------------------------------------ async def search( self, query: str, k: int = 8, memory_type: MemoryType | None = None, filters: dict | None = None, ) -> list[dict]: embedding = self._embedder.embed(query) collections_to_search = ( [self._collections[memory_type]] if memory_type else [ self._collections[MemoryType.EPISODIC], self._collections[MemoryType.SEMANTIC], self._collections[MemoryType.PROCEDURAL], ] ) results: list[dict] = [] loop = asyncio.get_event_loop() for col in collections_to_search: where = filters or None r = await loop.run_in_executor( None, lambda c=col, w=where: c.query( query_embeddings=[embedding], n_results=min(k, max(1, c.count())), where=w, include=["documents", "metadatas", "distances"], ), ) for doc, meta, dist in zip( r["documents"][0], r["metadatas"][0], r["distances"][0] ): results.append({"text": doc, "metadata": meta, "score": 1.0 - dist}) # Blend recency + semantic score now = time.time() for r in results: age_days = (now - r["metadata"].get("timestamp", now)) / 86400 decay = 0.5 ** (age_days / DECAY_HALF_LIFE_DAYS) r["weighted_score"] = r["score"] * decay * r["metadata"].get("importance", 1.0) results.sort(key=lambda x: x["weighted_score"], reverse=True) return results[:k] async def get_recent( self, memory_type: MemoryType, n: int = 50 ) -> list[dict]: col = self._collections[memory_type] loop = asyncio.get_event_loop() r = await loop.run_in_executor( None, lambda: col.get(include=["documents", "metadatas"]), ) paired = sorted( zip(r["documents"], r["metadatas"], r["ids"]), key=lambda x: x[1].get("timestamp", 0), reverse=True, ) return [{"text": d, "metadata": m, "id": i} for d, m, i in paired[:n]] async def update_importance(self, mem_id: str, memory_type: MemoryType, importance: float): col = self._collections[memory_type] loop = asyncio.get_event_loop() existing = await loop.run_in_executor( None, lambda: col.get(ids=[mem_id], include=["metadatas", "documents"]) ) if not existing["ids"]: return meta = existing["metadatas"][0] meta["importance"] = importance await loop.run_in_executor( None, lambda: col.update(ids=[mem_id], metadatas=[meta]), ) async def promote(self, mem_id: str, text: str, metadata: dict): """Move a memory from episodic → semantic (consolidation step).""" await self.add( text=text, memory_type=MemoryType.SEMANTIC, metadata={**metadata, "promoted": True} ) col = self._collections[MemoryType.EPISODIC] loop = asyncio.get_event_loop() await loop.run_in_executor(None, lambda: col.delete(ids=[mem_id])) 

6. cognitive_core/dreamer.py

""" DreamEngine — runs offline consolidation when the robot is docked. Cycle: 1. Experience Replay — surface recent failures for re-weighting 2. Schema Induction — cluster similar episodes → semantic facts 3. Failure Analysis — extract preconditions from failed actions 4. Memory Promotion — move high-importance episodic → semantic 5. Synthetic Imagination — generate counterfactual "what-if" memories 6. Forgetting — decay & prune low-importance old memories """ from __future__ import annotations import asyncio import logging import time from .memory_store import MemoryStore, MemoryType from .embedder import Embedder logger = logging.getLogger(__name__) _IMPORTANCE_PRUNE_THRESHOLD = 0.05 _PROMOTION_IMPORTANCE_THRESHOLD = 0.7 class DreamEngine: def __init__(self, memory: MemoryStore, embedder: Embedder): self._memory = memory self._embedder = embedder self._abort_event = asyncio.Event() self._running = False async def abort(self): if self._running: logger.warning("DreamEngine: abort requested mid-cycle.") self._abort_event.set() async def run_full_cycle(self): self._abort_event.clear() self._running = True t0 = time.monotonic() logger.info("=== Dream Cycle START ===") steps = [ ("Experience Replay", self._experience_replay), ("Schema Induction", self._schema_induction), ("Failure Analysis", self._failure_analysis), ("Memory Promotion", self._memory_promotion), ("Synthetic Imagination", self._synthetic_imagination), ("Forgetting Pass", self._forgetting_pass), ] for name, fn in steps: if self._abort_event.is_set(): logger.warning(f"Dream Cycle aborted before: {name}") break try: logger.info(f" [{name}] ...") await fn() except Exception as exc: logger.error(f" [{name}] error: {exc}", exc_info=True) elapsed = time.monotonic() - t0 self._running = False logger.info(f"=== Dream Cycle END ({elapsed:.1f}s) ===") # ------------------------------------------------------------------ # Step 1 — Experience replay # ------------------------------------------------------------------ async def _experience_replay(self): """Boost importance of recent failure memories.""" recent = await self._memory.get_recent(MemoryType.EPISODIC, n=100) for item in recent: meta = item["metadata"] if not meta.get("success", True): current = meta.get("importance", 1.0) boosted = min(current * 1.5, 2.0) await self._memory.update_importance( item["id"], MemoryType.EPISODIC, boosted ) # ------------------------------------------------------------------ # Step 2 — Schema induction # ------------------------------------------------------------------ async def _schema_induction(self): """ Cluster recent episodic memories by embedding similarity. When ≥3 similar memories exist, write a generalised semantic fact. """ recent = await self._memory.get_recent(MemoryType.EPISODIC, n=200) if len(recent) < 3: return texts = [m["text"] for m in recent] embeddings = self._embedder.embed_batch(texts) import numpy as np # local import to avoid cold-start cost emb_array = np.array(embeddings) # Simple greedy clustering (cosine similarity threshold 0.85) used = [False] * len(texts) for i, (text_i, emb_i) in enumerate(zip(texts, emb_array)): if used[i]: continue cluster_texts = [text_i] cluster_idx = [i] for j, (text_j, emb_j) in enumerate(zip(texts, emb_array)): if i == j or used[j]: continue sim = float(np.dot(emb_i, emb_j)) if sim >= 0.85: cluster_texts.append(text_j) cluster_idx.append(j) if len(cluster_texts) >= 3: schema = f"[SCHEMA] Pattern observed across {len(cluster_texts)} episodes: {cluster_texts[0][:120]}..." await self._memory.add( text=schema, memory_type=MemoryType.SEMANTIC, metadata={"source": "schema_induction", "cluster_size": len(cluster_texts)}, ) for idx in cluster_idx: used[idx] = True logger.info(f" Induced schema from {len(cluster_texts)} episodes.") # ------------------------------------------------------------------ # Step 3 — Failure analysis # ------------------------------------------------------------------ async def _failure_analysis(self): """Promote failure patterns to procedural memory as preconditions.""" recent = await self._memory.get_recent(MemoryType.EPISODIC, n=100) failures = [m for m in recent if not m["metadata"].get("success", True)] for item in failures: text = item["text"] precondition = ( f"[PRECONDITION] Before attempting this action, " f"verify conditions to avoid: {text[:200]}" ) existing = await self._memory.search( query=precondition, k=1, memory_type=MemoryType.PROCEDURAL, ) if not existing or existing[0]["score"] < 0.92: await self._memory.add( text=precondition, memory_type=MemoryType.PROCEDURAL, metadata={ "source": "failure_analysis", "original_episode": text[:100], "importance": 1.5, }, ) # ------------------------------------------------------------------ # Step 4 — Memory promotion # ------------------------------------------------------------------ async def _memory_promotion(self): """Move high-importance episodic memories to semantic store.""" recent = await self._memory.get_recent(MemoryType.EPISODIC, n=200) for item in recent: if item["metadata"].get("importance", 1.0) >= _PROMOTION_IMPORTANCE_THRESHOLD: await self._memory.promote( mem_id=item["id"], text=item["text"], metadata={**item["metadata"], "promoted_at": time.time()}, ) # ------------------------------------------------------------------ # Step 5 — Synthetic imagination # ------------------------------------------------------------------ async def _synthetic_imagination(self): """ Generate counterfactual memories: "What if the opposite had been true?" These seed the semantic store with alternative strategies for reflection. """ failures = await self._memory.search( query="action failed error", k=5, memory_type=MemoryType.EPISODIC, filters={"success": False}, ) for item in failures: counterfactual = ( f"[IMAGINATION] If the preconditions had been met, " f"the action might have succeeded: {item['text'][:150]}" ) await self._memory.add( text=counterfactual, memory_type=MemoryType.SEMANTIC, metadata={"source": "imagination", "importance": 0.6}, ) # ------------------------------------------------------------------ # Step 6 — Forgetting # ------------------------------------------------------------------ async def _forgetting_pass(self): """ Prune low-importance episodic memories using time-decayed importance. Semantic and procedural memories are NOT pruned automatically. """ import numpy as np recent = await self._memory.get_recent(MemoryType.EPISODIC, n=500) now = time.time() to_delete: list[str] = [] col = self._memory._collections[MemoryType.EPISODIC] for item in recent: age_days = (now - item["metadata"].get("timestamp", now)) / 86400 decay = 0.5 ** (age_days / 30.0) effective = item["metadata"].get("importance", 1.0) * decay if effective < _IMPORTANCE_PRUNE_THRESHOLD: to_delete.append(item["id"]) if to_delete: loop = asyncio.get_event_loop() await loop.run_in_executor(None, lambda: col.delete(ids=to_delete)) logger.info(f" Pruned {len(to_delete)} decayed episodic memories.") 

7. cognitive_core/world_model.py

""" WorldModel — lightweight dynamic representation of the robot's environment. Tracks: - Current perceptual state (latest VLM caption) - Active task + goal - Recently observed object/surface types (for novelty detection) - Known failure contexts """ from __future__ import annotations import time from dataclasses import dataclass, field @dataclass class WorldModel: current_perception: str = "" active_task: str = "" active_goal: str = "" known_surfaces: set[str] = field(default_factory=set) known_objects: set[str] = field(default_factory=set) recent_failure_contexts: list[str] = field(default_factory=list) last_updated: float = field(default_factory=time.time) def update_perception(self, caption: str): self.current_perception = caption self.last_updated = time.time() # Naive surface/object extraction from VLM text for word in caption.lower().split(): if word in {"smooth", "rough", "metal", "plastic", "wood", "glass", "fabric"}: self.known_surfaces.add(word) def record_failure(self, context: str): self.recent_failure_contexts.append(context) if len(self.recent_failure_contexts) > 20: self.recent_failure_contexts.pop(0) def novel_surfaces(self, known_set: set[str]) -> set[str]: return self.known_surfaces - known_set def to_dict(self) -> dict: return { "perception": self.current_perception, "task": self.active_task, "goal": self.active_goal, "surfaces": list(self.known_surfaces), "failures": self.recent_failure_contexts[-5:], } 

8. cognitive_core/reflection.py

""" ReflectionEngine — meta-cognitive self-analysis and curiosity-driven goal generation. Uses the local LLM (via local_llm.py) to: - Analyse recent failures and extract lessons - Generate exploration goals based on novelty detection - Periodically ask "which strategies work best for me?" """ from __future__ import annotations import logging from .memory_store import MemoryStore, MemoryType from .world_model import WorldModel logger = logging.getLogger(__name__) class ReflectionEngine: def __init__(self, memory: MemoryStore): self._memory = memory # Lazy import to avoid circular dependency self._llm = None def _get_llm(self): if self._llm is None: from local_llm import LocalLLM self._llm = LocalLLM() return self._llm async def analyse(self, context: dict) -> str: """ Given a failure/success context, produce a short lesson string and store it in procedural memory. """ llm = self._get_llm() prompt = ( f"You are the robot's self-reflection module.\n" f"Context: {context}\n" f"In one sentence: what went wrong and how should the robot adapt next time?" ) lesson = await llm.generate(prompt, max_tokens=80) logger.info(f"Reflection lesson: {lesson}") await self._memory.add( text=f"[REFLECTION] {lesson}", memory_type=MemoryType.PROCEDURAL, metadata={"source": "reflection", "importance": 1.2}, ) return lesson async def curiosity_goal(self, world_model: WorldModel) -> str | None: """ Return an exploration goal if novel surfaces or objects are detected. Goal is chosen to reduce uncertainty. """ # Retrieve surfaces the robot has prior procedural knowledge about known_results = await self._memory.search( query="surface interaction knowledge", k=20, memory_type=MemoryType.PROCEDURAL, ) known_surface_texts = " ".join(r["text"] for r in known_results).lower() novel = { s for s in world_model.known_surfaces if s not in known_surface_texts } if novel: goal = f"Explore and learn about interaction with {', '.join(novel)} surfaces." logger.info(f"Curiosity goal generated: {goal}") return goal # Check if any recent failures have no procedural entry if world_model.recent_failure_contexts: last_fail = world_model.recent_failure_contexts[-1] existing = await self._memory.search( query=last_fail, k=1, memory_type=MemoryType.PROCEDURAL ) if not existing or existing[0]["score"] < 0.8: return f"Investigate failure pattern: {last_fail[:80]}" return None async def meta_strategy_review(self) -> str: """ Periodic review: 'Which action strategies have the highest success rate?' Writes a semantic summary memory. """ procedural = await self._memory.get_recent(MemoryType.PROCEDURAL, n=50) if not procedural: return "Insufficient procedural memory for meta-strategy review." llm = self._get_llm() memories_text = "\n".join(f"- {m['text'][:100]}" for m in procedural[:20]) prompt = ( f"You are the robot's meta-cognition module.\n" f"Here are the robot's procedural memories:\n{memories_text}\n\n" f"Summarise in 2 sentences: which strategies work best and what to avoid." ) summary = await llm.generate(prompt, max_tokens=100) await self._memory.add( text=f"[META-STRATEGY] {summary}", memory_type=MemoryType.SEMANTIC, metadata={"source": "meta_review", "importance": 1.3}, ) logger.info(f"Meta-strategy review: {summary}") return summary 

9. local_llm.py

""" LocalLLM — wrapper around Ollama (default) with optional cloud fallback. Ollama must be running locally: `ollama serve` Default model: phi3:mini (fast, ~2GB RAM). For docking-station inference with GPU, swap to: llama3:8b, qwen2:7b, gemma2:9b """ from __future__ import annotations import asyncio import logging import os logger = logging.getLogger(__name__) _DEFAULT_MODEL = os.getenv("DAVID_LLM_MODEL", "phi3:mini") _OLLAMA_URL = os.getenv("DAVID_OLLAMA_URL", "http://localhost:11434") class LocalLLM: def __init__(self, model: str = _DEFAULT_MODEL): self.model = model self._client = None def _get_client(self): if self._client is None: import ollama self._client = ollama.AsyncClient(host=_OLLAMA_URL) return self._client async def generate(self, prompt: str, max_tokens: int = 200) -> str: """Generate a completion for `prompt` using the local model.""" client = self._get_client() try: response = await client.generate( model=self.model, prompt=prompt, options={"num_predict": max_tokens, "temperature": 0.3}, ) return response["response"].strip() except Exception as exc: logger.error(f"LocalLLM error ({self.model}): {exc}") # Return a safe fallback rather than crashing the node return "[LLM unavailable]" async def chat(self, messages: list[dict], max_tokens: int = 200) -> str: """Chat completion interface (role/content format).""" client = self._get_client() try: response = await client.chat( model=self.model, messages=messages, options={"num_predict": max_tokens, "temperature": 0.3}, ) return response["message"]["content"].strip() except Exception as exc: logger.error(f"LocalLLM chat error: {exc}") return "[LLM unavailable]" 

10. nodes/cognitive_node.py

""" CognitiveNode — the central mind of David v3.0. Subscribes to: /perception/vlm_caption (String, SensorData QoS) /control/action_outcome (TaskOutcome, Reliable QoS) Publishes: /cognition/internal_state (String, Reliable) — world model JSON /cognition/goal (String, Reliable) — active curiosity goal Runs: - Continuous micro-reflection after every failure - Periodic meta-strategy review (every 30 minutes) - Curiosity-goal generation on novel percepts """ import asyncio import json import time import rclpy from rclpy.node import Node from rclpy.qos import QoSPresetProfiles from std_msgs.msg import String from cognitive_core import CognitiveCore from cognitive_core.memory_store import MemoryType # Replace 'your_package' with your actual ROS 2 package name # from your_package.msg import TaskOutcome REFLECTION_COOLDOWN_SECONDS = 10.0 META_REVIEW_INTERVAL_SECONDS = 1800.0 # 30 minutes class CognitiveNode(Node): """ Central cognition node. Maintains the WorldModel and runs self-reflection + goal generation continuously. """ def __init__(self, core: CognitiveCore): super().__init__("cognitive_node") self._core = core self._last_reflection_time = 0.0 self._last_meta_review_time = 0.0 reliable = QoSPresetProfiles.SERVICES_DEFAULT.value sensor = QoSPresetProfiles.SENSOR_DATA.value self.vlm_sub = self.create_subscription( String, "/perception/vlm_caption", self._on_perception, sensor ) self.outcome_sub = self.create_subscription( String, "/control/action_outcome", self._on_outcome, reliable ) self.state_pub = self.create_publisher(String, "/cognition/internal_state", reliable) self.goal_pub = self.create_publisher(String, "/cognition/goal", reliable) # Periodic meta-review timer self.create_timer(META_REVIEW_INTERVAL_SECONDS, self._timer_meta_review) # Publish world model snapshot at 1 Hz self.create_timer(1.0, self._publish_state) self.get_logger().info("CognitiveNode started — mind is online.") # ------------------------------------------------------------------ # Perception # ------------------------------------------------------------------ def _on_perception(self, msg: String): asyncio.ensure_future(self._handle_perception(msg.data)) async def _handle_perception(self, caption: str): self._core.world_model.update_perception(caption) goal = await self._core.generate_goal() if goal: self._core.world_model.active_goal = goal out = String() out.data = goal self.goal_pub.publish(out) self.get_logger().info(f"[Curiosity] New goal: {goal}") # Store percept in working memory await self._core.store( text=f"PERCEPT: {caption}", memory_type=MemoryType.WORKING ) # ------------------------------------------------------------------ # Action outcomes # ------------------------------------------------------------------ def _on_outcome(self, msg: String): asyncio.ensure_future(self._handle_outcome(msg.data)) async def _handle_outcome(self, outcome_json: str): try: data = json.loads(outcome_json) except json.JSONDecodeError: data = {"action": outcome_json, "success": True, "error": ""} success = data.get("success", True) action = data.get("action", "unknown") if not success: self._core.world_model.record_failure(f"{action}: {data.get('error', '')}") now = time.monotonic() if (now - self._last_reflection_time) >= REFLECTION_COOLDOWN_SECONDS: self._last_reflection_time = now context = { "action": action, "error": data.get("error", ""), "perception": self._core.world_model.current_perception, } lesson = await self._core.reflect(context) self.get_logger().info(f"[Reflection] {lesson}") # ------------------------------------------------------------------ # Periodic timers # ------------------------------------------------------------------ def _timer_meta_review(self): asyncio.ensure_future(self._run_meta_review()) async def _run_meta_review(self): summary = await self._core.reflection.meta_strategy_review() self.get_logger().info(f"[Meta-Review] {summary}") def _publish_state(self): state = json.dumps(self._core.world_model.to_dict()) msg = String() msg.data = state self.state_pub.publish(msg) 

11. nodes/narrator_node.py

""" NarratorNode — translates VLM captions into episodic memories. Unchanged filtering logic from v2.0; memory backend is now CognitiveCore (local). """ import asyncio import time import rclpy from rclpy.node import Node from rclpy.qos import QoSPresetProfiles from std_msgs.msg import String from cognitive_core import CognitiveCore from cognitive_core.memory_store import MemoryType RATE_LIMIT_SECONDS = 3.0 class NarratorNode(Node): def __init__(self, core: CognitiveCore): super().__init__("narrator_node") self._core = core self._last_observation = "" self._last_sent_time = 0.0 self.subscription = self.create_subscription( String, "/perception/vlm_caption", self.vlm_callback, QoSPresetProfiles.SENSOR_DATA.value, ) self.get_logger().info("NarratorNode started (SensorData QoS, local memory).") def vlm_callback(self, msg: String): current = msg.data now = time.monotonic() if current == self._last_observation: return if (now - self._last_sent_time) < RATE_LIMIT_SECONDS: return self._last_observation = current self._last_sent_time = now self.get_logger().info(f"Observing: {current}") asyncio.ensure_future(self._store(current)) async def _store(self, observation: str): try: await self._core.store( text=observation, memory_type=MemoryType.EPISODIC, metadata={"source": "vlm"}, ) except Exception as exc: self.get_logger().error(f"NarratorNode store error: {exc}") 

12. nodes/feedback_node.py

""" FeedbackNode — captures typed TaskOutcome messages and stores causal memories. Now writes to EPISODIC (raw) and optionally triggers a quick reflection via CognitiveCore. """ import asyncio import json import rclpy from rclpy.node import Node from rclpy.qos import QoSPresetProfiles from std_msgs.msg import String from cognitive_core import CognitiveCore from cognitive_core.memory_store import MemoryType # In production use typed msg: from your_package.msg import TaskOutcome class FeedbackNode(Node): def __init__(self, core: CognitiveCore): super().__init__("feedback_node") self._core = core self.subscription = self.create_subscription( String, "/control/action_outcome", self.outcome_callback, QoSPresetProfiles.SERVICES_DEFAULT.value, ) self.get_logger().info("FeedbackNode started (Reliable QoS, local memory).") def outcome_callback(self, msg: String): asyncio.ensure_future(self._handle(msg.data)) async def _handle(self, raw: str): try: data = json.loads(raw) action = data.get("action", raw) success = data.get("success", True) error = data.get("error", "") except json.JSONDecodeError: action, success, error = raw, True, "" outcome_str = "SUCCEEDED" if success else "FAILED" memory_text = f"ACTION: {action} | OUTCOME: {outcome_str} | ERROR: {error or 'none'}" self.get_logger().info(f"Feedback ingested: {memory_text}") try: await self._core.store( text=memory_text, memory_type=MemoryType.EPISODIC, metadata={"causal": True, "success": success}, ) except Exception as exc: self.get_logger().error(f"FeedbackNode store error: {exc}") 

13. nodes/planner_node.py

""" PlannerNode — enriches high-level commands with retrieved memory context. Now queries both EPISODIC and PROCEDURAL stores and includes active goal context. Token budget preserved from v2.0. """ import asyncio import rclpy from rclpy.node import Node from rclpy.qos import QoSPresetProfiles from std_msgs.msg import String from cognitive_core import CognitiveCore from cognitive_core.memory_store import MemoryType MAX_HISTORY_CHARS = 2000 class PlannerNode(Node): def __init__(self, core: CognitiveCore): super().__init__("planner_node") self._core = core reliable = QoSPresetProfiles.SERVICES_DEFAULT.value self.command_sub = self.create_subscription( String, "/command/high_level", self.command_callback, reliable ) self.context_pub = self.create_publisher( String, "/command/contextualized", reliable ) self.get_logger().info("PlannerNode started (Reliable QoS, local retrieval).") def command_callback(self, msg: String): asyncio.ensure_future(self._enrich(msg.data)) async def _enrich(self, task: str): self.get_logger().info(f"Enriching task: '{task}'") episodic = await self._core.retrieve( query=f"how to {task} and past failures", k=5, memory_type=MemoryType.EPISODIC, ) procedural = await self._core.retrieve( query=f"preconditions and skills for {task}", k=5, memory_type=MemoryType.PROCEDURAL, ) all_memories = episodic + procedural relevant: list[str] = [] chars = 0 for mem in all_memories: text = mem["text"] if chars + len(text) > MAX_HISTORY_CHARS: self.get_logger().warn("Memory budget reached — truncating.") break relevant.append(text) chars += len(text) # Include active curiosity goal if it overlaps with the task goal_context = "" active_goal = self._core.world_model.active_goal if active_goal and any(word in task.lower() for word in active_goal.lower().split()): goal_context = f"ACTIVE GOAL: {active_goal}\n" prompt = f"TASK: {task}\n{goal_context}" if relevant: prompt += "RELEVANT MEMORY:\n - " + "\n - ".join(relevant[:6]) self.get_logger().info(f"Context prompt ready ({len(prompt)} chars).") out = String() out.data = prompt self.context_pub.publish(out) 

14. nodes/system_monitor.py

""" SystemMonitorNode — dock/undock detection + dream lifecycle management. Now calls CognitiveCore.consolidate() (local) instead of HTTP abort API. Abort is via in-process asyncio event — instant, no network. """ import asyncio import rclpy from rclpy.node import Node from rclpy.qos import QoSPresetProfiles from std_msgs.msg import String from cognitive_core import CognitiveCore class SystemMonitorNode(Node): def __init__(self, core: CognitiveCore): super().__init__("system_monitor") self._core = core self._is_sleeping = False self._dream_task: asyncio.Task | None = None self.battery_sub = self.create_subscription( String, "/hardware/battery_state", self.battery_callback, QoSPresetProfiles.SERVICES_DEFAULT.value, ) self.get_logger().info("SystemMonitorNode started (Reliable QoS, local dreaming).") def battery_callback(self, msg: String): asyncio.ensure_future(self._handle_battery(msg.data)) async def _handle_battery(self, state: str): if state == "CHARGING" and not self._is_sleeping: self._is_sleeping = True self.get_logger().info("Docked. Starting Dream Cycle...") self._dream_task = asyncio.create_task(self._core.consolidate()) elif state == "DISCHARGING" and self._is_sleeping: self._is_sleeping = False self.get_logger().info("Undocked. Aborting dream if active...") await self._core.abort_consolidation() if self._dream_task and not self._dream_task.done(): self._dream_task.cancel() try: await self._dream_task except asyncio.CancelledError: self.get_logger().info("Dream task cancelled cleanly.") self._dream_task = None 

15. launch_cognitive.py

""" Entry point for David v3.0. Starts the local CognitiveCore then spins all ROS nodes. Configuration via ROS parameters (or env vars as fallback). """ import asyncio import json import threading import time import rclpy from rclpy.executors import MultiThreadedExecutor from std_msgs.msg import String from cognitive_core import CognitiveCore from nodes.cognitive_node import CognitiveNode from nodes.narrator_node import NarratorNode from nodes.feedback_node import FeedbackNode from nodes.planner_node import PlannerNode from nodes.system_monitor import SystemMonitorNode def _simulate_robot_life(sim_node): """Demo publisher sequence for development/testing.""" pub_vlm = sim_node.create_publisher(String, "/perception/vlm_caption", 10) pub_outcome = sim_node.create_publisher(String, "/control/action_outcome", 10) pub_cmd = sim_node.create_publisher(String, "/command/high_level", 10) pub_battery = sim_node.create_publisher(String, "/hardware/battery_state", 10) time.sleep(2) print("\n--- SIMULATION START (David v3.0) ---") # Novel surface percept → triggers curiosity goal pub_vlm.publish(String(data="approached workstation, surface is smooth_glass")) time.sleep(1) pub_vlm.publish(String(data="approached workstation, surface is smooth_glass")) time.sleep(4) # Task attempt → failure → micro-reflection pub_cmd.publish(String(data="grasp fragile glass beaker")) time.sleep(2) outcome = json.dumps({ "action": "grasp glass beaker", "success": False, "error": "shattered due to excessive force on smooth_glass", }) pub_outcome.publish(String(data=outcome)) time.sleep(3) # Dock → local dream cycle pub_battery.publish(String(data="CHARGING")) time.sleep(4) # Early undock → in-process abort (no HTTP needed) pub_battery.publish(String(data="DISCHARGING")) time.sleep(2) # Retry with enriched procedural context pub_cmd.publish(String(data="grasp fragile glass beaker")) async def async_main(): rclpy.init() # ------------------------------------------------------------------ # Read config from ROS parameters # ------------------------------------------------------------------ param_node = rclpy.create_node("david_param_node") param_node.declare_parameter("memory_dir", "/var/david/memory") param_node.declare_parameter("llm_model", "phi3:mini") param_node.declare_parameter("robot_id", "David_v3") memory_dir = param_node.get_parameter("memory_dir").get_parameter_value().string_value llm_model = param_node.get_parameter("llm_model").get_parameter_value().string_value robot_id = param_node.get_parameter("robot_id").get_parameter_value().string_value param_node.destroy_node() import os os.environ["DAVID_LLM_MODEL"] = llm_model print(f"[{robot_id}] Initialising local CognitiveCore at {memory_dir}...") core = CognitiveCore(persist_dir=memory_dir) await core.initialise() print(f"[{robot_id}] CognitiveCore ready. Spinning ROS nodes...") # Instantiate all nodes cognitive = CognitiveNode(core) narrator = NarratorNode(core) feedback = FeedbackNode(core) planner = PlannerNode(core) monitor = SystemMonitorNode(core) sim_node = rclpy.create_node("simulator_node") executor = MultiThreadedExecutor() for n in (cognitive, narrator, feedback, planner, monitor): executor.add_node(n) threading.Thread( target=_simulate_robot_life, args=(sim_node,), daemon=True ).start() try: loop = asyncio.get_event_loop() await loop.run_in_executor(None, executor.spin) except KeyboardInterrupt: pass finally: executor.shutdown() for n in (cognitive, narrator, feedback, planner, monitor, sim_node): n.destroy_node() rclpy.shutdown() print(f"[{robot_id}] Shutdown complete.") def main(): asyncio.run(async_main()) if __name__ == "__main__": main()

How to Run

1. Install Ollama and pull the default model

curl -fsSL https://ollama.com/install.sh | sh ollama pull phi3:mini # ~2 GB — fast on-robot model # Optional: for docking-station inference # ollama pull llama3:8b 

2. Install Python dependencies

pip install chromadb sentence-transformers ollama aiohttp pydantic numpy 

3. Build the ROS 2 package

cd ~/ros2_ws colcon build --packages-select robot_david source install/setup.bash 

4. Launch

# With defaults (phi3:mini, /var/david/memory): python3 launch_cognitive.py # With ROS 2 launch overrides: ros2 launch robot_david cognitive.launch.py # Override model at runtime: ros2 launch robot_david cognitive.launch.py \ llm_model:=llama3:8b \ memory_dir:=/data/david/persistent 

5. Inspect memory at runtime

# Watch the world model ros2 topic echo /cognition/internal_state # Watch curiosity goals ros2 topic echo /cognition/goal # Watch enriched command context ros2 topic echo /command/contextualized
