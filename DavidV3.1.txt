6. cognitive_core/dreamer.py

Changes: (a) _experience_replay now sorts by surprise_score — high-surprise failures get the biggest importance boost. (b) New run_micro_cycle() for lightweight 10-minute naps. (c) _abort_event shared between both tiers.

""" DreamEngine — offline consolidation engine for David v3.1. MACRO cycle (run_full_cycle — docked only): 1. Surprise-Weighted Experience Replay 2. Schema Induction 3. Failure Analysis 4. 1.0) boosted_val = min(current * factor, 2.0) await self._memory.update_importance( item["id"], MemoryType.EPISODIC, boosted_val ) boosted += 1 logger.info(f" Quick re-weight applied to {boosted} surprise failures.") # ------------------------------------------------------------------ # MACRO CYCLE — "Deep Sleep" (Improvement #4) # ------------------------------------------------------------------ async def run_full_cycle(self): """Full consolidation — run only when docked (GPU-intensive).""" self._abort_event.clear() self._running = True t0 = time.monotonic() logger.info("=== Dream Cycle (Deep Sleep) START ===") steps = [ ("Surprise-Weighted Experience Replay", self._experience_replay), ("Schema Induction", self._schema_induction), ("Failure Analysis", self._failure_analysis), ("Memory Promotion", self._memory_promotion), ("Synthetic Imagination", self._synthetic_imagination), ("Forgetting Pass", self._forgetting_pass), ] for name, fn in steps: if self._abort_event.is_set(): logger.warning(f"Dream Cycle aborted before: {name}") break try: logger.info(f" [{name}] ...") await fn() except Exception as exc: logger.error(f" [{name}] error: {exc}", exc_info=True) elapsed = time.monotonic() - 7. cognitive_core/world_model.py

Changes: Replaces naive string matching with an embedding centroid approach. calculate_surprise() computes cosine distance from a running centroid of recent perceptual embeddings. The update_perception() method now optionally accepts and tracks the embedding.

""" WorldModel — dynamic representation of the robot's environment. v3.1 changes (Improvement #1 — Embedding-Based Novelty Detection): - Maintains a rolling centroid of perception embeddings. - calculate_surprise() computes cosine distance from that centroid instead of naive word-matching. - update_perception() now accepts an optional pre-computed embedding so the CognitiveNode can pass it in without recomputing. """ self.current_perception = caption self.last_updated = time.time() # Legacy surface extraction (kept as a secondary signal) for word in caption.lower().split(): if word in {"smooth", "rough", "metal", "plastic", "wood", "glass", "fabric"}: self.known_surfaces.add(word) # Update rolling centroid if embedding is not None: vec = np.array(embedding, dtype=np.float32) self._perception_embeddings.append(vec) if len(self._perception_embeddings) > _CENTROID_WINDOW: self._perception_embeddings.pop(0) self._centroid = None # invalidate cached centroid # ------------------------------------------------------------------ # Surprise detection (Improvement #1) # ------------------------------------------------------------------ def _get_centroid(self) -> np.ndarray | None: """Return (and cache) the mean of the embedding window.""" if self._centroid is not None: return self._centroid if not self._perception_embeddings: return None centroid = np.mean(self._perception_embeddings, axis=0) norm = np.linalg.norm(centroid) if norm > 1e-8: centroid = centroid / norm self._centroid = centroid return self._centroid def surprise_from_embedding(self, embedding: list[float]) -> float: """ Compute a surprise score in [0, 1] for a new percept embedding.
