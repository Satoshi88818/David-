## 3. cognitive_core/dreamer.py (Stability & Stability-Aware Dreaming)
```python
"""
dreamer.py v4.5 — Humanoid Stability Simulation
Checks for both Physical Collisions and Fall Risks (CoM outside Support Polygon).
"""
import numpy as np
import logging
from cognitive_core.schemas import DreamerResult

logger = logging.getLogger(__name__)

class DreamerLoop:
    def __init__(self):
        self.support_polygon_radius = 0.15 # Approx footprint area

    def simulate(self, current_depth, proposed_primitive, world_model) -> DreamerResult:
        """Simulates full body movement and returns safety metrics."""
        
        # 1. Collision Check (Fast Path)
        collision_prob = self._check_collisions(world_model, proposed_primitive)
        
        # 2. Stability Check (NEW v4.5)
        fall_risk = self._check_stability(proposed_primitive, world_model)
        
        safe_vector = None
        safe_base = None
        
        if fall_risk > 0.4:
            logger.warning("Stability Alert: Fall risk detected. Calculating base adjustment.")
            # Suggest moving the base to re-center the CoM
            safe_base = [0.1, 0.0, 0.0] # Heuristic: step toward reach
            
        return DreamerResult(
            simulated=True,
            collision_prob=collision_prob,
            fall_risk=fall_risk,
            safe_vector=safe_vector,
            safe_base_adjustment=safe_base
        )

    def _check_stability(self, primitive, world_model) -> float:
        """Simplified Centroidal Dynamics check."""
        if not primitive.right_hand_xyz and not primitive.left_hand_xyz:
            return 0.0
            
        target = primitive.right_hand_xyz or primitive.left_hand_xyz
        # If target reach > 0.6m from body center, fall risk increases
        dist_from_center = np.linalg.norm([target[0], target[1]])
        
        if dist_from_center > 0.7: return 0.9 # Extreme lean
        if dist_from_center > 0.5: return 0.5 # Moderate lean
        return 0.1

    def _check_collisions(self, wm, prim) -> float:
        # Standard voxel sweep from v4.0
        return 0.0 
```

## 4. cognitive_core/social_grid.py (Social Navigation)

"""
social_grid.py v4.5 — Humanoid Path Yielding
Projects trajectories into both manipulation and locomotion space.
"""
from cognitive_core.schemas import HumanAgent, SocialOccupancyPatch
import time
import numpy as np

class SocialGrid:
    def __init__(self):
        self.agents: Dict[str, HumanAgent] = {}

    def update(self, vla_agents: List[HumanAgent]):
        for a in vla_agents:
            self.agents[a.agent_id] = a
            
    def get_navigation_constraints(self) -> List[SocialOccupancyPatch]:
        """Returns repulsive patches for the locomotion planner."""
        patches = []
        for agent in self.agents.values():
            # Create a 0.6m personal space cylinder
            patches.append(SocialOccupancyPatch(
                agent_id=agent.agent_id,
                centre_xyz=agent.position_xyz,
                radius_m=0.8, # Humanoid needs wider berth
                repulsion=0.9
            ))
        return patches

### Key Architectural Refinements in v4.5:

1.  **Stability First:** The `DreamerLoop` is no longer just a collision checker; it's a balance simulator. If David "dreams" he will fall, the `PlannerNode` intercepts the command and forces a leg movement first.
2.  **Bimanual Awareness:** Primitives now accept `left_hand` and `right_hand` targets simultaneously, allowing the VLA to coordinate complex tasks like "Hold the bowl with left while whisking with right."
3.  **Humanoid Diagnosis:** The `DiagnosticReasoner` is aware of `BALANCE_LIMIT`. Instead of David just giving up when he can't reach a high shelf, he reasons: *"I am too short/unstable. I must walk closer or find a step stool."*
4.  **Social Locomotion:** The `SocialGrid` creates "repulsion zones" that the humanoid base must navigate around, preventing David from bumping into people while he walks toward a goal.
